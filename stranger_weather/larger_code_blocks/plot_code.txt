path_prefix=''
maindf=pd.read_csv(f'{path_prefix}/filename')
maindf=maindf[(maindf['year']>=1995)&(maindf['year']<=2019)]
major_counties = {6412: 'Frankfurt',3241: 'Hannover',

counties_list=list(maindf['Kreise_Code'].unique())

demap=gpd.read_file(f'{path_prefix}\\official_de_shapefiles.gk3.shape.ebenen\\vg5000_ebenen_1231\\VG5000_KRS.shp',crs=4326)
demap['Kreise_Code']=demap['ARS'].astype(int)
demap=demap.to_crs(epsg=3034)
demap=demap[['Kreise_Code','geometry']]
class map_counties_climate:
    
    def __init__(self,demap=demap,major_counties=major_counties):
        self.demap=demap
        self.major_counties = major_counties
        self.city_labels=[]
        for code, name in self.major_counties.items():
            county = self.demap[self.demap['Kreise_Code'] == code]
            if not county.empty:
                centroid = county.geometry.centroid.iloc[0]
                self.city_labels.append((name, centroid.x, centroid.y))
        
    def mapping_counties_extremes(self,plot_df,plot_subtitles=[],plot_title='',excess_flag=''):
        self.plot_df=plot_df
        self.plot_subtitles=plot_subtitles
        self.merged_df = self.demap.merge(self.plot_df.reset_index(), on='Kreise_Code')
        self.merged_df=self.merged_df.dropna()
        plot_columns = self.plot_df.columns
        
        right_cols = plot_columns[0:2]
        left_cols = plot_columns[2:4]
        #as_high_low = (len(high_cols)>0) or (len(low_cols)>0)
        
        vmin_right=self.merged_df[right_cols].min().min() #if high_cols else None
        vmax_right =self.merged_df[right_cols].max().max() #if high_cols else None
    
        vmin_left = self.merged_df[left_cols].min().min() #if low_cols else None
        vmax_left = self.merged_df[left_cols].max().max() #if low_cols else None
    
        fig, axes = plt.subplots(figsize=(22, 8), nrows=1, ncols=4)
        
        for ix, plot_column in enumerate(plot_columns):
            ax = axes[ix]
            if ('is_high' in plot_column) and ('tas' in plot_column):
                cmap_right = 'Reds'
                cmap = cmap_right
                vmin, vmax = vmin_right, vmax_right
                
            elif ('how_high' in plot_column) and ('tas' in plot_column):
                cmap_left = 'Reds'
                cmap = cmap_left
                vmin, vmax = vmin_left, vmax_left
            
            elif ('is_low' in plot_column) and ('tas' in plot_column):
                cmap_right = 'Greens'
                cmap = cmap_right
                vmin, vmax = vmin_right, vmax_right
                
            elif ('how_low' in plot_column) and ('tas' in plot_column):
                cmap_left = 'Greens'
                cmap = cmap_left
                vmin, vmax = vmin_left, vmax_left
            
            elif ('is_high' in plot_column) and ('prec' in plot_column):
                cmap_right = 'Blues'
                cmap = cmap_right
                vmin, vmax = vmin_right, vmax_right
                
            elif ('how_high' in plot_column) and ('prec' in plot_column):
                cmap_left = 'Blues'
                cmap = cmap_left
                vmin, vmax = vmin_left, vmax_left
                
            elif ('is_low' in plot_column) and ('prec' in plot_column):
                cmap_right = 'Oranges'
                cmap=cmap_right
                vmin, vmax = vmin_right, vmax_right
                
            elif ('how_low' in plot_column) and ('prec' in plot_column):
                cmap_left = 'Oranges'
                cmap=cmap_left
                vmin, vmax = vmin_left, vmax_left
                
            self.merged_df.plot(column=plot_column,ax=ax,legend=False,cmap=cmap,edgecolor='lightgrey',linewidth=0.07,
                           vmin=vmin,vmax=vmax,missing_kwds={'facecolor': 'lightgrey'})
            
            ax.set_title(plot_subtitles[ix], fontsize=13, fontweight='bold', pad=10)
            ax.axis('off')
            
            for name, x, y in self.city_labels:
                ax.annotate(name,xy=(x, y),fontsize=8,fontweight='bold',ha='center',
                            bbox=dict(boxstyle='round,pad=0.25',facecolor='white',
                                      edgecolor='black',alpha=0.85))

        if len(right_cols)>0:
            norm_right = mcolors.Normalize(vmin=vmin_right, vmax=vmax_right)
            sm_right = cm.ScalarMappable(cmap=cmap_right, norm=norm_right)
            sm_right._A = []

            # Calculate position spanning axes[0] and axes[1]
            pos0 = axes[0].get_position()
            pos1 = axes[1].get_position()
            cbar_ax_right = fig.add_axes([pos0.x0, 0.08, pos1.x1 - pos0.x0, 0.025])
            fig.colorbar(sm_right, cax=cbar_ax_right, orientation='horizontal', label='Extremes Count')

    # Add shared horizontal colorbar for Lows (under subplots 2 and 3)
        if len(left_cols)>0:
            norm_left = mcolors.Normalize(vmin=vmin_left, vmax=vmax_left)
            sm_left = cm.ScalarMappable(cmap=cmap_left, norm=norm_left)
            sm_left._A = []

            # Calculate position spanning axes[2] and axes[3]
            pos2 = axes[2].get_position()
            pos3 = axes[3].get_position()
            cbar_ax_left = fig.add_axes([pos2.x0, 0.08, pos3.x1 - pos2.x0, 0.025])
            fig.colorbar(sm_left, cax=cbar_ax_left, orientation='horizontal', label=f'Extremes Excess {excess_flag}')
        
        plt.suptitle(plot_title,fontsize=15)

        plt.subplots_adjust(bottom=0.18, wspace=0.05)
        plt.show()
        
        
    def mapping_counties_moments(self,plot_df,plot_subtitles=[],title_text=''):
        self.plot_df=plot_df
        self.plot_subtitles=plot_subtitles
        self.merged_df = self.demap.merge(self.plot_df.reset_index(), on='Kreise_Code')
        plot_columns = self.plot_df.columns
        
        fig, axes = plt.subplots(figsize=(22, 8), nrows=1, ncols=len(plot_subtitles))
        for ix, plot_column in enumerate(plot_columns):
            ax = axes[ix]
            if 'mean_tas' in plot_column:
                cmap='Reds'
            elif 'stdev_tas' in plot_column:
                cmap='RdPu'
            elif 'mean_prec' in plot_column:
                cmap='Blues'
            elif 'stdev_prec' in plot_column:
                cmap='GnBu'
            vmin,vmax=self.merged_df[plot_column].min(),self.merged_df[plot_column].max()
            self.merged_df.plot(column=plot_column,ax=ax,legend=False,cmap=cmap,edgecolor='k',linewidth=0.3,
                           vmin=vmin,vmax=vmax,missing_kwds={'facecolor': 'lightgrey'})
            
            ax.set_title(plot_subtitles[ix], fontsize=13, fontweight='bold', pad=10)
            ax.axis('off')
            
            for name, x, y in self.city_labels:
                ax.annotate(name,xy=(x, y),fontsize=8,fontweight='bold',ha='center',
                            bbox=dict(boxstyle='round,pad=0.25',facecolor='white',
                                      edgecolor='black',alpha=0.85))
                
            norm = mcolors.Normalize(vmin=vmin, vmax=vmax)
            sm = cm.ScalarMappable(cmap=cmap, norm=norm)
            sm._A = []  # Dummy array required for ScalarMappable

            # shrink=0.65 keeps the colorbar shorter than the full width of the map
            # pad=0.04 controls vertical distance below the subplot
            cbar = fig.colorbar(sm, ax=ax, orientation='horizontal', shrink=0.65, pad=0.04, aspect=15)
            cbar.ax.tick_params(labelsize=9)
                
        plt.subplots_adjust(bottom=0.18, wspace=0.05)
        plt.show()
        
    def mapping_counties_moment(self, plot_df, plot_subtitles=[], colors='RdPu',figsize=(22,24),vmin=False,vmax=False):

        self.plot_df = plot_df
        self.plot_subtitles = plot_subtitles
        self.merged_df = self.demap.merge(self.plot_df.reset_index(), on='Kreise_Code')
        plot_columns = self.plot_df.columns
        
        if vmin:
            vmin=vmin
        else:
            vmin = self.merged_df[plot_columns].min().min()
        if vmax:
            vmax=vmax
        else:
            vmax = self.merged_df[plot_columns].max().max()

        fig, axes = plt.subplots(figsize=figsize, nrows=1, ncols=len(plot_subtitles))
        for ix, plot_column in enumerate(plot_columns):
            ax = axes[ix]
            self.merged_df.plot(column=plot_column, ax=ax, legend=False, cmap=colors,
                                 edgecolor='lightgrey', linewidth=0.029,
                                 vmin=vmin, vmax=vmax,
                                 missing_kwds={'facecolor': 'black'})

            ax.set_title(plot_subtitles[ix], fontsize=figsize[0]-3, fontweight='bold', pad=10)
            ax.axis('off')

            for name, x, y in self.city_labels:
                ax.annotate(name, xy=(x, y), fontsize=8, fontweight='bold', ha='center',
                            bbox=dict(boxstyle='round,pad=0.25', facecolor='white',
                                      edgecolor='black', alpha=0.30))

        norm = mcolors.Normalize(vmin=vmin, vmax=vmax)
        sm = cm.ScalarMappable(cmap=colors, norm=norm)
        sm._A = []  # dummy array required for ScalarMappable

        # attach colorbar explicitly to all four axes so it spans/centers correctly
        cbar = fig.colorbar(sm, ax=axes.tolist(), orientation='horizontal',
                             shrink=0.65, pad=0.08, aspect=30)
        cbar.ax.tick_params(labelsize=9)

        plt.show()
        
    def periodly_time_series(self, plot_df, plot_col, period_flag,title_text='',figsize=(22,24)):
        major_counties = list(self.major_counties.keys())
        self.plot_df = plot_df[plot_df['Kreise_Code'].isin(major_counties)].copy()

        if period_flag == 'month':
            period_range = range(1, 13)
            nrows, ncols = 3, 4
            period_labels = ['Jan','Feb','Mar','Apr','May','Jun',
                              'Jul','Aug','Sep','Oct','Nov','Dec']
        elif period_flag == 'season':
            period_range = range(1, 5)
            nrows, ncols = 2, 2
            period_labels = ['Winter', 'Spring', 'Summer', 'Autumn']
        else:
            raise ValueError("period_flag must be 'month' or 'season'")

        fig, axes = plt.subplots(nrows=nrows, ncols=ncols, figsize=figsize)
        axes = axes.flatten()  # so we can index with a single integer regardless of grid shape

        for period in period_range:
            ax = axes[period - 1]
            for county in major_counties:
                temp_df = self.plot_df[
                    (self.plot_df[period_flag] == period) &
                    (self.plot_df['Kreise_Code'] == county)
                ][['year', plot_col]].copy()
                temp_df = temp_df.set_index('year').sort_index()

                ax.plot(temp_df.index, temp_df[plot_col], marker='o', markersize=3,
                        label=self.major_counties[county])

            ax.set_title(period_labels[period - 1], fontsize=16, fontweight='bold')
            ax.set_xlabel('')
            ax.tick_params(axis='x', rotation=45, labelsize=12)
            ax.tick_params(axis='y', labelsize=16)

        # turn off any unused subplot axes (e.g. if grid has more slots than periods)
        for extra_ax in axes[len(list(period_range)):]:
            extra_ax.axis('off')

        # single shared legend, built from the last populated axis's handles
        handles, labels = axes[list(period_range)[-1] - 1].get_legend_handles_labels()
        fig.legend(handles, labels, loc='lower center', ncol=min(len(major_counties), 6),
                   bbox_to_anchor=(0.5, -0.06), fontsize=18)

        fig.suptitle(title_text, fontsize=18, fontweight='bold', y=1.01)
        plt.tight_layout()
        plt.show()

def make_descriptive_tables(table,round_level=2):
    ncols,nrows=table.shape[1],table.shape[0]
    feature_names={'mean':'mean','std':'std. dev',"min":'min','max':'max'}
    print('\\begin{table}[ht]')
    print('  \\centering')
    print('  \\caption{WRITE CAPTION}')
    print('  \\label{tab:write_label}')
    
    # Column setup: Left align first column, center the rest
    col_config = 'l' + 'c' * ncols
    print('  \\begin{tabular}{' + col_config + '}')
    print('    \\toprule')
    
    header = 'Variable'
    for i in range(ncols):
        header += f' & ({i+1})'
    print(f'    {header} \\\\')
    print('    \\midrule')
    for feature in ['mean','std','min','max']:
        line=feature_names[feature]
        for value in table.loc[feature].values:
            value=str(np.round(value,round_level))
            line=line+'&'+value
        print(line+'\\\\')
